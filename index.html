import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; function json(body: unknown, status = 200) { return new Response(JSON.stringify(body), { status, headers: { ...corsHeaders, "Content-Type": "application/json; charset=utf-8", }, }); } Deno.serve(async (req: Request) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } if (req.method !== "POST") { return json( { error: "روش درخواست پشتیبانی نمی‌شود." }, 405, ); } const supabaseUrl = Deno.env.get("SUPABASE_URL"); const serviceRoleKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); const adminEmail = (Deno.env.get("ADMIN_EMAIL") || "") .trim() .toLowerCase(); if (!supabaseUrl || !serviceRoleKey || !adminEmail) { return json( { error: "تنظیمات سمت سرور کامل نیست." }, 500, ); } const authorization = req.headers.get("Authorization") || ""; const token = authorization.replace(/^Bearer\s+/i, "").trim(); if (!token) { return json( { error: "ابتدا وارد حساب مدیر شوید." }, 401, ); } const adminClient = createClient( supabaseUrl, serviceRoleKey, { auth: { persistSession: false, autoRefreshToken: false, }, }, ); // بررسی هویت کاربر با توکن معتبر Supabase const { data: authData, error: authError } = await adminClient.auth.getUser(token); const caller = authData?.user; // فقط حساب مدیر مجاز به استفاده از این API است if ( authError || !caller?.email || caller.email.trim().toLowerCase() !== adminEmail ) { return json( { error: "دسترسی فقط برای مدیر مجاز است." }, 403, ); } let body: Record; try { body = await req.json(); } catch { return json( { error: "درخواست نامعتبر است." }, 400, ); } const action = String(body.action || ""); const now = new Date().toISOString(); // دریافت اطلاعات داشبورد و فهرست کاربران if (action === "dashboard" || action === "users") { const { data: authDataList, error: listError } = await adminClient.auth.admin.listUsers({ page: 1, perPage: 1000, }); if (listError) { return json( { error: "دریافت کاربران ناموفق بود." }, 500, ); } // نام ستون‌ها مطابق ساختار واقعی جدول profiles است const { data: profiles, error: profileError } = await adminClient .from("profiles") .select( "id, username, mobile, created_at, trial_started_at, trial_ends_at, subscription_ends_at, subscription_status", ); if (profileError) { return json( { error: "دریافت اطلاعات کاربران ناموفق بود. ساختار جدول profiles را بررسی کنید.", }, 500, ); } const profileMap = new Map( (profiles || []).map((profile: any) => [ profile.id, profile, ]), ); const users = (authDataList?.users || []).map((authUser: any) => { const profile: any = profileMap.get(authUser.id) || {}; const trialEnd = profile.trial_ends_at ? new Date(profile.trial_ends_at).getTime() : 0; const subscriptionEnd = profile.subscription_ends_at ? new Date(profile.subscription_ends_at).getTime() : 0; let status = profile.subscription_status || "trial"; if (subscriptionEnd > Date.now()) { status = "active"; } else if (trialEnd > Date.now()) { status = "trial"; } else if (status !== "active") { status = "expired"; } return { id: authUser.id, email: authUser.email || "", username: profile.username || "", // ستون واقعی دیتابیس mobile است، نه phone mobile: profile.mobile || "", created_at: authUser.created_at || profile.created_at || null, trial_started_at: profile.trial_started_at || null, trial_ends_at: profile.trial_ends_at || null, subscription_ends_at: profile.subscription_ends_at || null, subscription_status: status, }; }); const counts = { total: users.length, active: users.filter( (user: any) => user.subscription_status === "active", ).length, trial: users.filter( (user: any) => user.subscription_status === "trial", ).length, expired: users.filter( (user: any) => user.subscription_status === "expired", ).length, }; return json({ ok: true, users, counts, }); } // تغییر وضعیت اشتراک کاربر if (action === "update") { const userId = String(body.user_id || "").trim(); const status = String(body.status || "").trim(); const days = Number(body.days ?? 30); if ( !userId || !["trial", "active", "expired"].includes(status) ) { return json( { error: "اطلاعات کاربر یا وضعیت اشتراک معتبر نیست.", }, 400, ); } if ( !Number.isFinite(days) || days < 1 || days > 3650 ) { return json( { error: "مدت باید بین ۱ تا ۳۶۵۰ روز باشد." }, 400, ); } // مطمئن شو پروفایل کاربر وجود دارد const { data: existingProfile, error: findError } = await adminClient .from("profiles") .select("id") .eq("id", userId) .maybeSingle(); if (findError) { return json( { error: "بررسی اطلاعات پروفایل ناموفق بود." }, 500, ); } if (!existingProfile) { return json( { error: "پروفایل این کاربر در جدول profiles وجود ندارد.", }, 404, ); } const endDate = new Date( Date.now() + days * 86400000, ).toISOString(); const patch: Record = { subscription_status: status, }; if (status === "active") { patch.subscription_ends_at = endDate; } if (status === "trial") { patch.trial_started_at = now; patch.trial_ends_at = endDate; patch.subscription_ends_at = null; } if (status === "expired") { patch.subscription_ends_at = now; } const { error: updateError } = await adminClient .from("profiles") .update(patch) .eq("id", userId); if (updateError) { return json( { error: "ذخیره تغییرات اشتراک ناموفق بود." }, 500, ); } // ثبت تاریخچه عملیات مدیر const { error: auditError } = await adminClient .from("admin_audit_logs") .insert({ admin_email: caller.email, target_user_id: userId, action: status === "active" ? "set_subscription" : status === "trial" ? "reset_trial" : "expire_subscription", details: { status, days, ends_at: status === "expired" ? now : endDate, }, }); if (auditError) { console.error( "Admin audit log insert failed:", auditError.message, ); } return json({ ok: true, message: "تغییرات اشتراک ذخیره شد.", }); } // دریافت تاریخچه عملیات مدیر if (action === "history") { const { data, error } = await adminClient .from("admin_audit_logs") .select( "id, admin_email, target_user_id, action, details, created_at", ) .order("created_at", { ascending: false }) .limit(100); if (error) { return json( { error: "دریافت تاریخچه عملیات ناموفق بود." }, 500, ); } return json({ ok: true, history: data || [], }); } return json( { error: "عملیات ناشناخته است." }, 400, ); });
